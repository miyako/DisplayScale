cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(display-scale LANGUAGES CXX)

add_executable(display-scale main.cpp)
target_compile_features(display-scale PRIVATE cxx_std_17)
set_target_properties(display-scale PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(display-scale PRIVATE /utf-8)
elseif(MINGW)
  target_link_options(display-scale PRIVATE -municode)
endif()
if(APPLE)
  target_link_libraries(display-scale PRIVATE "-framework CoreGraphics" "-framework CoreFoundation")
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]")
    message(FATAL_ERROR "DisplayScale supports native Windows x64 only")
  endif()
  target_compile_definitions(display-scale PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(display-scale PRIVATE user32 shcore)
  if(MSVC)
    target_link_options(display-scale PRIVATE
      "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/display-scale.manifest")
  else()
    enable_language(RC)
    target_sources(display-scale PRIVATE display-scale.rc)
    target_include_directories(display-scale PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(display-scale-unit-tests tests.cpp)
  add_executable(display-scale-test-echo test_echo.cpp)
  foreach(target display-scale-unit-tests display-scale-test-echo)
    target_compile_features(${target} PRIVATE cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8)
    elseif(MINGW AND target STREQUAL "display-scale-test-echo")
      target_link_options(${target} PRIVATE -municode)
    endif()
    if(APPLE)
      target_link_libraries(${target} PRIVATE "-framework CoreGraphics" "-framework CoreFoundation")
    endif()
    if(WIN32)
      target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
      target_link_libraries(${target} PRIVATE user32 shcore)
    endif()
  endforeach()
  add_test(NAME display-scale-pure-backend COMMAND display-scale-unit-tests)
endif()
